"""Publish a tiny safetensors checkpoint through the real owner loader."""

import hashlib
import json
import tempfile
from pathlib import Path

import torch
from safetensors.torch import save_file

from vllm.models.deepseek_v41.nvidia.ops.shared_engram_owner import (
    publish_checkpoint,
)
from vllm.models.deepseek_v41.nvidia.ops.shared_engram_vmm import import_table


def main() -> None:
    torch.cuda.set_device(0)
    with tempfile.TemporaryDirectory() as temporary:
        model = Path(temporary)
        config = {
            "text_config": {
                "engram_layer_ids": [1],
                "engram_num_embeddings": [3],
                "engram_max_ngram_size": 2,
                "engram_vocab_size": 3,
                "engram_n_heads": 1,
                "engram_head_dim": 64,
                "engram_pad_token_id": 0,
                "engram_compressed_vocab_size": 3,
            }
        }
        (model / "config.json").write_text(json.dumps(config))
        weight_key = "layers.1.engram.embed.weight"
        scale_key = "layers.1.engram.embed.scale"
        weights = torch.ones((3, 64), dtype=torch.float32).to(torch.float8_e4m3fn)
        weights[-1] = 2
        scales = torch.full((3, 2), 127, dtype=torch.uint8).view(
            torch.float8_e8m0fnu
        )
        save_file(
            {weight_key: weights, scale_key: scales},
            model / "model-00001-of-00001.safetensors",
        )
        index = {
            "weight_map": {
                weight_key: "model-00001-of-00001.safetensors",
                scale_key: "model-00001-of-00001.safetensors",
            }
        }
        (model / "model.safetensors.index.json").write_text(json.dumps(index))
        manifest, mappings = publish_checkpoint(model, 0)
        imported = None
        try:
            assert len(mappings) == 1
            mapping = mappings[0]
            assert mapping.cpu is not None
            expected = torch.cat(
                (weights.view(torch.uint8).flatten(), scales.view(torch.uint8).flatten())
            )
            torch.testing.assert_close(mapping.cpu, expected, rtol=0, atol=0)
            assert manifest["tables"]["1"]["weight_sha256"] == hashlib.sha256(
                weights.view(torch.uint8).numpy().tobytes()
            ).hexdigest()
            imported = import_table(mapping.table, 0)
            torch.testing.assert_close(imported.cuda.cpu(), expected, rtol=0, atol=0)
            print(
                json.dumps(
                    {
                        "result": "pass",
                        "generation": manifest["generation"],
                        "rows": manifest["tables"]["1"]["rows"],
                        "head_sizes": manifest["tables"]["1"]["head_sizes"],
                    }
                )
            )
        finally:
            if imported is not None:
                imported.close()
            for mapping in reversed(mappings):
                mapping.close()


if __name__ == "__main__":
    main()
