"""Separate-process FABRIC import and production Engram gather smoke."""

import argparse
import json
import subprocess
import sys
import tempfile
import threading
from pathlib import Path

import torch

from vllm.models.deepseek_v41.common import engram as engram_ops
from vllm.models.deepseek_v41.common.engram import _engram_lookup_kernel
from vllm.models.deepseek_v41.nvidia.ops.shared_engram_owner import (
    _OwnerServer,
    _OwnerState,
    attach_table,
    detach_table,
)
from vllm.models.deepseek_v41.nvidia.ops.shared_engram_vmm import create_table

ROWS = 65536
DIM = 256
SCALE_WIDTH = DIM // 32
INDICES = (0, 1, 257, ROWS // 2, ROWS - 1)


def gather(full: torch.Tensor, indices: torch.Tensor, out: torch.Tensor) -> None:
    weight_bytes = ROWS * DIM
    weight = full[:weight_bytes].view(torch.float8_e4m3fn).view(ROWS, DIM)
    scales = full[weight_bytes:].view(ROWS, SCALE_WIDTH)
    _engram_lookup_kernel[(1,)](
        weight,
        scales,
        indices,
        indices,
        out,
        0,
        ROWS,
        len(INDICES),
        1,
        1,
        HEAD_START=0,
        LOCAL_HEADS=1,
        TOTAL_HEADS=1,
        DIM=DIM,
        QUANT_BLOCK=32,
        BLOCK_R=16,
        GRID=1,
        SORTED=False,
        PACKED=False,
        OUT_STRIDE=DIM,
    )


def consumer(endpoint: str, token_file: Path) -> None:
    torch.cuda.set_device(1)
    connection, mapping, manifest = attach_table(
        endpoint, token_file, 1, ROWS, DIM, (ROWS,), 1
    )
    try:
        indices = torch.tensor(INDICES, dtype=torch.int32, device="cuda")
        out = torch.empty((len(INDICES), DIM), dtype=torch.bfloat16, device="cuda")
        gather(mapping.cuda, indices, out)
        torch.cuda.synchronize()
        expected = torch.ones_like(out)
        expected[-1] = 2
        torch.testing.assert_close(out, expected, rtol=0, atol=0)
        graph = torch.cuda.CUDAGraph()
        with torch.cuda.graph(graph):
            gather(mapping.cuda, indices, out)
        graph.replay()
        torch.cuda.synchronize()
        torch.testing.assert_close(out, expected, rtol=0, atol=0)
        del graph
        print(json.dumps({"result": "pass", "imported_va": mapping.address}))
    finally:
        detach_table(connection, mapping, manifest["lease"])

    # Exercise the model's opt-in storage path too. Its atexit release keeps
    # the mapping live through all graph objects until this worker exits.
    engram_ops.get_engram_dp_size = lambda: 1
    engram_ops.get_tensor_model_parallel_world_size = lambda: 1
    engram_ops.get_tensor_model_parallel_rank = lambda: 0
    embedding = engram_ops.ParallelEngramEmbedding(
        ROWS,
        DIM,
        (ROWS,),
        cpu_offload=True,
        shared_host_endpoint=endpoint,
        shared_host_token_file=str(token_file),
        shared_host_generation="test-generation",
        layer_id=1,
    )
    assert embedding.weight.is_meta and embedding.weight_scale_inv.is_meta
    indices_2d = indices[:, None]
    embedded = torch.empty((len(INDICES), 1, DIM), dtype=torch.bfloat16, device="cuda")
    embedding.lookup(indices_2d, embedded)
    torch.cuda.synchronize()
    torch.testing.assert_close(embedded[:, 0], expected, rtol=0, atol=0)
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        embedding.lookup(indices_2d, embedded)
    graph.replay()
    torch.cuda.synchronize()
    torch.testing.assert_close(embedded[:, 0], expected, rtol=0, atol=0)
    del graph


def owner() -> None:
    torch.cuda.set_device(0)
    mapping = create_table(ROWS * (DIM + SCALE_WIDTH), 0)
    try:
        assert mapping.cpu is not None
        weight_bytes = ROWS * DIM
        values = torch.ones((ROWS, DIM), dtype=torch.float8_e4m3fn)
        values[-1] = 2
        mapping.cpu[:weight_bytes].copy_(values.view(torch.uint8).flatten())
        mapping.cpu[weight_bytes:].fill_(127)
        with tempfile.TemporaryDirectory() as temporary:
            token_file = Path(temporary) / "token"
            token = "0123456789abcdef0123456789abcdef"
            token_file.write_text(token)
            manifest = {
                "schema": 1,
                "generation": "test-generation",
                "format": "FP8_E4M3FN_E8M0_BLOCK32",
                "tables": {
                    "1": {
                        "rows": ROWS,
                        "dim": DIM,
                        "scale_width": SCALE_WIDTH,
                        "head_sizes": [ROWS],
                        "table": mapping.table.to_wire(),
                    }
                },
            }
            state = _OwnerState(manifest, token)
            with _OwnerServer(("127.0.0.1", 0), state) as server:
                thread = threading.Thread(target=server.serve_forever)
                thread.start()
                try:
                    endpoint = f"127.0.0.1:{server.server_address[1]}"
                    result = subprocess.run(
                        [
                            sys.executable,
                            str(Path(__file__).resolve()),
                            "--consumer",
                            "--endpoint",
                            endpoint,
                            "--token-file",
                            str(token_file),
                        ],
                        check=True,
                        capture_output=True,
                        text=True,
                    )
                    print(result.stdout.strip())
                    assert not state.active and not state.uncertain
                finally:
                    server.shutdown()
                    thread.join()
        print(
            json.dumps(
                {
                    "result": "pass",
                    "owner_va": mapping.address,
                    "backing_bytes": mapping.table.padded_bytes,
                    "fabric_handles": len(mapping.table.fabric_handles),
                }
            )
        )
    finally:
        mapping.close()


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--consumer", action="store_true")
    parser.add_argument("--endpoint")
    parser.add_argument("--token-file", type=Path)
    args = parser.parse_args()
    if args.consumer:
        consumer(args.endpoint, args.token_file)
    else:
        owner()
