"""Small CUDA VMM owner/import alias and graph replay check.

Run only while holding a lease for the selected physical GPU.
"""

import argparse
import json

import torch

from vllm.models.deepseek_v41.nvidia.ops.shared_engram_vmm import (
    create_table,
    import_table,
)


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--device", type=int, default=0)
    args = parser.parse_args()
    torch.cuda.set_device(args.device)
    owner = create_table(16 << 20, args.device)
    imported = None
    try:
        assert owner.cpu is not None
        owner.cpu[:1024].copy_(torch.arange(1024, dtype=torch.int32).to(torch.uint8))
        imported = import_table(owner.table, args.device)
        indices = torch.tensor([0, 1, 255, 256, 1023], device="cuda")
        expected = owner.cpu[indices.cpu()].cuda()
        actual = imported.cuda[indices]
        torch.testing.assert_close(actual, expected, rtol=0, atol=0)
        output = torch.empty_like(actual)
        graph = torch.cuda.CUDAGraph()
        with torch.cuda.graph(graph):
            output.copy_(imported.cuda[indices])
        graph.replay()
        torch.cuda.synchronize()
        torch.testing.assert_close(output, expected, rtol=0, atol=0)
        owner.cpu[0] = 73
        graph.replay()
        torch.cuda.synchronize()
        assert output[0].item() == 73
        del graph
        print(
            json.dumps(
                {
                    "result": "pass",
                    "device": args.device,
                    "numa_node": owner.table.numa_node,
                    "padded_bytes": owner.table.padded_bytes,
                    "fabric_handles": len(owner.table.fabric_handles),
                    "owner_va": owner.address,
                    "imported_va": imported.address,
                }
            )
        )
    finally:
        if imported is not None:
            imported.close()
        owner.close()


if __name__ == "__main__":
    main()
