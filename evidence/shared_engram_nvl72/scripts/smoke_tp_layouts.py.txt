"""TP1/TP2/TP4 consumers attach one immutable owner generation."""

import argparse
import hashlib
import json
import subprocess
import sys
import tempfile
import threading
from pathlib import Path
from types import SimpleNamespace

import torch

from vllm.models.deepseek_v41.common import engram as engram_ops
from vllm.models.deepseek_v41.nvidia.ops.shared_engram_owner import (
    _OwnerServer,
    _OwnerState,
)
from vllm.models.deepseek_v41.nvidia.ops.shared_engram_vmm import create_table

DIM = 256
HEAD_SIZES = tuple(97 + index for index in range(24))
ROWS = sum(HEAD_SIZES)
TOKENS = 5


def _hash(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def consumer(endpoint: str, token_file: Path, checkpoint_dir: Path, tp: int, rank: int):
    torch.cuda.set_device(rank)
    engram_ops.get_engram_dp_size = lambda: 1
    engram_ops.get_tensor_model_parallel_world_size = lambda: tp
    engram_ops.get_tensor_model_parallel_rank = lambda: rank
    engram_ops.get_current_vllm_config = lambda: SimpleNamespace(
        model_config=SimpleNamespace(model=str(checkpoint_dir), model_weights=None)
    )
    embedding = engram_ops.ParallelEngramEmbedding(
        ROWS,
        DIM,
        HEAD_SIZES,
        cpu_offload=True,
        shared_host_endpoint=endpoint,
        shared_host_token_file=str(token_file),
        shared_host_generation="one-generation",
        layer_id=1,
    )
    assert embedding.weight.is_meta and embedding.weight_scale_inv.is_meta
    offsets = [0]
    for size in HEAD_SIZES:
        offsets.append(offsets[-1] + size)
    ids = torch.tensor(
        [
            [offsets[head] + (token * 7 + head) % HEAD_SIZES[head] for head in range(24)]
            for token in range(TOKENS)
        ],
        dtype=torch.int32,
        device="cuda",
    )
    out = torch.empty(
        (TOKENS, embedding.part_n_hash_cols, DIM),
        dtype=torch.bfloat16,
        device="cuda",
    )
    expected = torch.tensor(
        [
            [
                [float(int(ids[token, head].item()) % 7 + 1)] * DIM
                for head in range(
                    embedding.head_start,
                    embedding.head_start + embedding.part_n_hash_cols,
                )
            ]
            for token in range(TOKENS)
        ],
        dtype=torch.bfloat16,
    )
    embedding.lookup(ids, out)
    torch.cuda.synchronize()
    torch.testing.assert_close(out.cpu(), expected, rtol=0, atol=0)
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        embedding.lookup(ids, out)
    graph.replay()
    torch.cuda.synchronize()
    torch.testing.assert_close(out.cpu(), expected, rtol=0, atol=0)
    del graph
    print(
        json.dumps(
            {
                "result": "pass",
                "tp": tp,
                "rank": rank,
                "head_start": embedding.head_start,
                "local_heads": embedding.part_n_hash_cols,
                "rows": embedding.part_num_embeddings,
            }
        ),
        flush=True,
    )


def owner() -> None:
    torch.cuda.set_device(0)
    mapping = create_table(ROWS * (DIM + DIM // 32), 0)
    try:
        assert mapping.cpu is not None
        weight = torch.tensor(
            [(row % 7) + 1 for row in range(ROWS)], dtype=torch.float32
        )[:, None].expand(ROWS, DIM)
        fp8 = weight.to(torch.float8_e4m3fn)
        weight_bytes = ROWS * DIM
        mapping.cpu[:weight_bytes].copy_(fp8.view(torch.uint8).flatten())
        mapping.cpu[weight_bytes:].fill_(127)
        with tempfile.TemporaryDirectory() as temporary:
            root = Path(temporary)
            token_file = root / "token"
            token_file.write_text("0123456789abcdef0123456789abcdef")
            (root / "config.json").write_text('{"generation":"one-generation"}')
            (root / "model.safetensors.index.json").write_text("{}")
            manifest = {
                "schema": 1,
                "generation": "one-generation",
                "config_sha256": _hash(root / "config.json"),
                "index_sha256": _hash(root / "model.safetensors.index.json"),
                "format": "FP8_E4M3FN_E8M0_BLOCK32",
                "tables": {
                    "1": {
                        "rows": ROWS,
                        "dim": DIM,
                        "scale_width": DIM // 32,
                        "head_sizes": HEAD_SIZES,
                        "table": mapping.table.to_wire(),
                    }
                },
            }
            state = _OwnerState(manifest, token_file.read_text())
            with _OwnerServer(("127.0.0.1", 0), state) as server:
                thread = threading.Thread(target=server.serve_forever)
                thread.start()
                endpoint = f"127.0.0.1:{server.server_address[1]}"
                try:
                    for tp in (1, 2, 4):
                        processes = [
                            subprocess.Popen(
                                [
                                    sys.executable,
                                    str(Path(__file__).resolve()),
                                    "--consumer",
                                    "--endpoint",
                                    endpoint,
                                    "--token-file",
                                    str(token_file),
                                    "--checkpoint-dir",
                                    str(root),
                                    "--tp",
                                    str(tp),
                                    "--rank",
                                    str(rank),
                                ],
                                stdout=subprocess.PIPE,
                                stderr=subprocess.STDOUT,
                                text=True,
                            )
                            for rank in range(tp)
                        ]
                        for process in processes:
                            stdout, _ = process.communicate(timeout=120)
                            print(stdout.strip(), flush=True)
                            if process.returncode:
                                raise RuntimeError(
                                    f"TP{tp} consumer failed: {process.returncode}"
                                )
                        with state.condition:
                            assert state.condition.wait_for(
                                lambda: not state.active, timeout=5
                            )
                            assert not state.uncertain
                    print(json.dumps({"result": "pass", "generation": "one-generation"}))
                finally:
                    server.shutdown()
                    thread.join()
    finally:
        mapping.close()


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--consumer", action="store_true")
    parser.add_argument("--endpoint")
    parser.add_argument("--token-file", type=Path)
    parser.add_argument("--checkpoint-dir", type=Path)
    parser.add_argument("--tp", type=int)
    parser.add_argument("--rank", type=int)
    args = parser.parse_args()
    if args.consumer:
        consumer(
            args.endpoint, args.token_file, args.checkpoint_dir, args.tp, args.rank
        )
    else:
        owner()
