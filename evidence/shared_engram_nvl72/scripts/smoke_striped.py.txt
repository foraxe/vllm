"""Verify a contiguous table VA backed by chunks on both Grace NUMA nodes."""

import json

import torch

from vllm.models.deepseek_v41.nvidia.ops.shared_engram_vmm import (
    create_table,
    import_table,
)


def main() -> None:
    torch.cuda.set_device(0)
    owner = create_table(
        4 << 20,
        0,
        numa_nodes=(0, 1),
        chunk_bytes=2 << 20,
        chunk_placement=(0, 1),
    )
    imported = None
    try:
        assert owner.table.placement == (0, 1)
        assert owner.cpu is not None
        positions = (0, (2 << 20) - 1, 2 << 20, (4 << 20) - 1)
        for index, position in enumerate(positions):
            owner.cpu[position] = 10 + index
        torch.cuda.set_device(1)
        imported = import_table(owner.table, 1)
        indices = torch.tensor(positions, device="cuda")
        actual = imported.cuda[indices]
        expected = torch.tensor([10, 11, 12, 13], dtype=torch.uint8, device="cuda")
        torch.testing.assert_close(actual, expected, rtol=0, atol=0)
        output = torch.empty_like(actual)
        graph = torch.cuda.CUDAGraph()
        with torch.cuda.graph(graph):
            output.copy_(imported.cuda[indices])
        graph.replay()
        torch.cuda.synchronize()
        torch.testing.assert_close(output, expected, rtol=0, atol=0)
        del graph
        print(
            json.dumps(
                {
                    "result": "pass",
                    "placement": owner.table.placement,
                    "owner_va": owner.address,
                    "imported_va": imported.address,
                }
            )
        )
    finally:
        if imported is not None:
            imported.close()
        owner.close()


if __name__ == "__main__":
    main()
